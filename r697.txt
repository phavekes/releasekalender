Voorbereiding
De diensten die nu een uid of eppn ontvangen van eduID worden op de uitzonderingslijst feature_force_global_uid_entities geplaatst.
Deze zijn in manage te vinden met deze mongo query in de manage database:



 db.saml20_idp.aggregate([
   {
      $match: {
        "data.entityid": "https://login.eduid.nl"
      }
   },
   {
      $unwind: "$data.allowedEntities"
   },
   {
      $facet: {
        saml20_sp: [
           {
              $lookup: {
                from: "saml20_sp",
                let: {
                   name: "$data.allowedEntities.name"
                },
                pipeline: [
                   {
                     $match: {
                       $expr: {
                         $and: [
                           { $eq: ["$data.entityid", "$$name"] },
                           {
                             $anyElementTrue: {
                                $map: {
                                  input: {
                                     $objectToArray: {
                                       $ifNull: ["$data.arp.attributes", {}]
                                     }
                                  },
                                  as: "attribute",
                                  in: {
                                     $or: [
                                       {
                                          $regexMatch: {
                                            input: "$$attribute.k",
                                            regex: "urn:mace:dir:attribute-def:eduPersonPrincipalName"
                                          }
                                       },
                                       {
                                          $regexMatch: {
                                            input: "$$attribute.k",
                                            regex: "urn:mace:dir:attribute-def:uid"
                                          }
                                       }
                                     ]
                                  }
                                }
                             }
                           }
                         ]
                       }
                     }
                   }
                ],
                as: "matches"
              }
           },
           { $unwind: "$matches" },
           {
              $project: {
                _id: 0,
                entityid: "$matches.data.entityid"
              }
           }
        ],
        oidc10_rp: [
           {
              $lookup: {
              from: "oidc10_rp",
              let: {
                 name: "$data.allowedEntities.name"
              },
              pipeline: [
                 {
                   $match: {
                     $expr: {
                       $and: [
                         { $eq: ["$data.entityid", "$$name"] },
                         {
                           $anyElementTrue: {
                              $map: {
                                input: {
                                   $objectToArray: {
                                     $ifNull: ["$data.arp.attributes", {}]
                                   }
                                },
                                as: "attribute",
                                in: {
                                   $or: [
                                     {
                                        $regexMatch: {
                                          input: "$$attribute.k",
                                          regex: "urn:mace:dir:attribute-def:eduPersonPrincipalName"
                                        }
                                     },
                                     {
                                        $regexMatch: {
                                          input: "$$attribute.k",
                                          regex: "urn:mace:dir:attribute-def:uid"
                                        }
                                     }
                                   ]
                                }
                              }
                           }
                         }
                       ]
                     }
                   }
                 }
              ],
              as: "matches"
             }
          },
          { $unwind: "$matches" },
          {
             $project: {
               _id: 0,
               entityid: "$matches.data.entityid"
             }
          }
      ]
  }
},
{
  $project: {
    entityids: {
      $concatArrays: [
        {
           $map: {
             input: "$saml20_sp",
             as: "service",
             in: "$$service.entityid"
           }
        },
        {
           $map: {
             input: "$oidc10_rp",
             as: "service",
             in: "$$service.entityid"
           }
        }
      ]
    }
      }
   },
   { $unwind: "$entityids" },
   {
      $group: {
        _id: "$entityids"
      }
   },
   { $sort: { _id: 1 } }

Deployment (The new way!)

  nr    Actie

 1     Go to OpenConext-deploy folder and checkout tag of OpenConext-deploy:

        cd ~/OpenConext-deploy
        git checkout 666




       Go to environments folder and check out tag of SURFconext-deploy-environments:

        cd ~/surfconext-depoyment-environments
        git checkout 666




 2     TEST

        cd ~/surfconext-depoyment-environments
        appdeploy test myconext




 3     ACC

        cd ~/surfconext-depoyment-environments

        blauw acc account 0 && blauw acc myconext 0 && blauw acc servicedesk 0
        appdeploy accblauw myconext --check
        blauw acc account 100 && blauw acc myconext 100 && blauw acc servicedesk 100
        appdeploy accrood myconext --check
        blauw acc account 50 && blauw acc myconext 50 && blauw acc servicedesk 50




 4     4e SET

        cd ~/surfconext-depoyment-environments

        appdeploy 4eset myconext --check




 5     PROD LB 100% rood, 0% blauw

        cd ~/surfconext-depoyment-environments
        blauw prd account 0 && blauw prd myconext 0 && blauw prd servicedesk 0
 6      PRODBLAUW




         cd ~/surfconext-depoyment-environments
         appdeploy prdblauw myconext --check




 7      PROD LB 0% rood, 100% blauw

         cd ~/surfconext-depoyment-environments
         blauw prd account 100 && blauw prd myconext 100 && blauw prd servicedesk 100




 8      PRODROOD




         cd ~/surfconext-depoyment-environments
         appdeploy prdrood myconext --check




 9      PROD LB 50/50

 10     Publieke website
